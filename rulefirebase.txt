rules_version = '2';

// ============================================================================
// Quy tắc bảo mật Firestore cho app "Nông nghiệp xã Bình Mỹ".
//
// TẠI SAO CẦN FILE NÀY: mọi kiểm tra quyền trong app.js (canManage, isAdmin,
// isSuperAdmin...) chỉ chạy trong trình duyệt — chỉ ẩn/hiện nút bấm, KHÔNG
// ngăn được ai đó mở DevTools, tự import Firebase SDK và gọi thẳng
// setDoc/deleteDoc để sửa dữ liệu hoặc tự phong mình làm admin. Quy tắc dưới
// đây là lớp chặn thật ở phía máy chủ Firestore, áp dụng bất kể request đến
// từ giao diện web hay từ console/script bất kỳ.
//
// CÁCH ÁP DỤNG: Firebase Console -> chọn dự án -> Firestore Database ->
// tab "Rules" -> dán toàn bộ nội dung file này -> Publish.
// ============================================================================

service cloud.firestore {
  match /databases/{database}/documents {

    function isSignedIn() {
      return request.auth != null;
    }
    // Đã là cán bộ (ward) hoặc quản trị viên xã (super) — có hồ sơ trong "admins".
    function isAdmin() {
      return isSignedIn()
        && exists(/databases/$(database)/documents/admins/$(request.auth.uid));
    }
    function isSuperAdmin() {
      return isSignedIn()
        && get(/databases/$(database)/documents/admins/$(request.auth.uid)).data.role == 'super';
    }

    // ---- admins/{uid}: tài khoản cán bộ ấp / quản trị viên xã ----
    match /admins/{uid} {
      // Tự tra vai trò của chính mình lúc đăng nhập (onAuthStateChanged, doPersonLogin).
      allow get: if isSignedIn() && request.auth.uid == uid;
      // Xem danh sách đầy đủ (màn "Quản lý tài khoản"): chỉ cán bộ mới xem được.
      allow list: if isAdmin();
      // Thêm/sửa/xoá tài khoản cán bộ: CHỈ Quản trị viên xã (role 'super').
      // -> Đây là chỗ chặn lỗ hổng nghiêm trọng nhất: nếu không có dòng này,
      //    bất kỳ ai tự đăng ký một tài khoản rồi tự setDoc vào admins/{uid_của_họ}
      //    với role:'super' là leo thang thành quản trị viên toàn quyền ngay lập tức.
      allow create, update, delete: if isSuperAdmin();
    }

    // ---- personUsers/{uid}: tài khoản hộ trồng / quán ăn-chợ ----
    match /personUsers/{uid} {
      allow get: if isSignedIn() && (request.auth.uid == uid || isAdmin());
      // Trước đây: getDocs(collection('personUsers')) được gọi cho MỌI khách xem trang
      // (kể cả chưa đăng nhập) chỉ để kiểm tra trùng email lúc đăng ký — vô tình lộ
      // số điện thoại/email của toàn bộ hộ trồng, quán ăn cho bất kỳ ai. Giờ chỉ admin
      // mới liệt kê được; việc kiểm tra trùng email đã có Firebase Auth tự chặn rồi
      // (báo lỗi "auth/email-already-in-use"), app.js không cần sửa gì thêm.
      allow list: if isAdmin();
      allow create: if isSignedIn() && request.auth.uid == uid;
      allow update, delete: if isSignedIn() && request.auth.uid == uid;
    }

    // ---- appData/{key}: seasons, households, qualityTests, outputs,
    //      procurements, products, activity — mỗi "bảng" là 1 document
    //      dạng { value: [ ...toàn bộ danh sách... ] }.
    match /appData/{key} {
      // Công khai cho mọi người xem, TRỪ "activity" (nhật ký thao tác nội bộ,
      // chỉ hiển thị trong màn Quản lý > Tổng quan nên chỉ admin cần đọc).
      allow read: if key != 'activity' || isAdmin();

      // seasons/households/qualityTests/outputs: khoá isAdmin().
      // procurements/products/activity: vẫn cho phép KHÔNG đăng nhập (thiết kế hiện tại)
      // nhưng GIỚI HẠN (interim, 22/09/2026):
      //  - value phải là list
      //  - không được làm list RÚT NGẮN (chặn wipe/xoá sạch)
      //  - không được tăng quá +10 phần tử / 1 lần ghi (chặn spam/lũ flooding)
      // Admin (isAdmin) bypass toàn bộ — import "replace" thu hẹp list vẫn chạy.
      // FIX TRIỆT để cần tách subcollection {key}/{itemId} — xem MIGRATION plan trong task.
      allow write: if (key in ['seasons', 'households', 'qualityTests', 'outputs'])
        ? isAdmin()
        : (isAdmin()
          || (request.resource.data.value is list
              && (resource == null || (
                request.resource.data.value.size() >= resource.data.value.size()
                && request.resource.data.value.size() <= resource.data.value.size() + 10
              ))));
    }
  }
}
